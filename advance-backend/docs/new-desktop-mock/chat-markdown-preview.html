How to wire a streaming chat endpoint

Below is a minimal WebSocket handler — drop it in src/http/chat.ts and you should see deltas flowing within ~120 ms of the first token.

typescript ⧉ Copy 
// src/http/chat.ts — append-only, hot-reloadable
import { streamText } from 'ai';
import { createDeepSeek } from '@ai-sdk/deepseek';

export async function handleChat(ws, body) {
  const stream = streamText({
    model: createDeepSeek().modelId('deepseek-chat'),
    messages: body.messages,
  });

  for await (const chunk of stream.fullStream) {
    if (chunk.type === 'text-delta') {
      ws.send(JSON.stringify({ type: 'text', data: chunk.text }));
    }
  }
}
